  options={[
    'The same, because the qty column must always be read',
    'The one with the filter asks for fewer, because DuckDB uses the min and max of each row group to skip the unrelated groups',
    'The one without the filter asks for fewer, because the filter needs extra reading',
    'It depends only on the speed of the network'
  ]}
  answer={1}

</Step>
