<Question

  client:visible
  q="A user sends the year 2026 (as a number) into a datetime field in lax mode. What do you get?"
  options={['An error, because it is not a date format', 'A date in the year 2026', '1970-01-01 00:33:46, because the number is read as unix seconds']}
  answer={2}
  why="Lax mode reads a number passed to datetime as a unix timestamp. 2026 seconds after the epoch is 00:33:46 on 1970-01-01. Use strict if you do not want this, since it rejects numbers (datetime_type)."
/>

</Step>
